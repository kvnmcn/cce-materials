    <h4 class="mb-3">Caesar Cipher</h4>

    <textarea id="textInput" class="form-control mb-2" rows="4" placeholder="Enter text..."></textarea>

    <input type="number" id="shiftInput" class="form-control mb-3" placeholder="Shift (e.g. 3)">

    <div class="d-flex gap-2 mb-3">
      <button class="btn btn-primary w-100" onclick="processText(true)">Encrypt</button>
      <button class="btn btn-secondary w-100" onclick="processText(false)">Decrypt</button>
    </div>

    <div id="errorMsg" class="text-danger"></div>
    <div id="result" class="mt-2"></div>

  </div>

  <script>
    function caesarCipher(str, shift) {
      const s = ((shift % 26) + 26) % 26;

      return str.split('').map(char => {
        const code = char.charCodeAt(0);

        if ((code >= 65 && code <= 90) || (code >= 97 && code <= 122)) {
          const base = code < 97 ? 65 : 97;
          return String.fromCharCode(((code - base + s) % 26) + base);
        }

        return char;
      }).join('');
    }

    function getInput() {
      const text = textInput.value.trim();
      const shift = Number(shiftInput.value);

      errorMsg.textContent = "";
      result.textContent = "";

      if (!text) return errorMsg.textContent = "Enter text.", null;
      if (shiftInput.value === "") return errorMsg.textContent = "Enter shift.", null;
      if (!Number.isInteger(shift)) return errorMsg.textContent = "Shift must be whole number.", null;

      return { text, shift };
    }

    function processText(isEncrypt) {
      const data = getInput();
      if (!data) return;

      const shift = isEncrypt ? data.shift : -data.shift;
      const output = caesarCipher(data.text, shift);

      result.textContent = (isEncrypt ? "Encrypted: " : "Decrypted: ") + output;
    }
  </script>
